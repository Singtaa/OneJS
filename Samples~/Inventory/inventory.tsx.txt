import { useRef, useState } from "react"
import { View, Text, useParticles, type ParticlesConfig, type VisualElement, type PointerEventData } from "onejs-react"

/**
 * A loot stash: the effect every ARPG, looter-shooter and gacha game ships.
 *
 *   Rarity aura   - one component, config driven entirely by the item's tier.
 *                   Rerolling changes the tier, and useParticles' deps recreate
 *                   the system from the new data.
 *   Legendary     - two emitters with different sprites in one system: soft glow
 *                   motes on the built-in disc, plus crisp shards on
 *                   Texture2D.whiteTexture. Emitters sharing a texture share a
 *                   draw call, so this costs exactly two.
 *   Drag trail    - space: "panel" keeps spawned particles pinned in panel space,
 *                   so the trail stays put instead of dragging along with the
 *                   element. One emitter per rarity, so no system is ever rebuilt
 *                   mid-drag.
 *
 * Note every particle system here lives on its own bare <View>. A host is
 * particle-owned - the system assigns style.unityMaterial to it, which would
 * cost the slot's rounded border its antialiasing.
 */

const COLS = 5
const ROWS = 3
const SLOT = 92
const GAP = 10
const GRID_W = COLS * SLOT + (COLS - 1) * GAP
const GRID_H = ROWS * SLOT + (ROWS - 1) * GAP

// Sharp square particles instead of the built-in soft disc. whiteTexture is
// opaque white, so it reads identically as premultiplied or straight alpha.
const WHITE = CS.UnityEngine.Texture2D.whiteTexture

export type Rarity = "common" | "uncommon" | "rare" | "epic" | "legendary"

const RARITY_ORDER: Rarity[] = ["common", "uncommon", "rare", "epic", "legendary"]

interface RarityDef {

}

const NAMES = [
    "Ashen Blade", "Tidecaller", "Moth Lantern", "Gravebloom", "Sunder Ring",
    "Vow of Cinders", "Pale Compass", "Riverstone", "Hollow Crown", "Emberweave",
    "Nightglass", "Warden's Seal", "Old Pact", "Saltwing", "Lastlight",
]

// Deterministic-ish spread so a fresh stash always looks varied.
function rollRarity(r: number): Rarity {
    if (r < 0.36) return "common"
    if (r < 0.64) return "uncommon"
    if (r < 0.84) return "rare"
    if (r < 0.955) return "epic"
    return "legendary"
}

interface Item {
    name: string
    rarity: Rarity
    power: number
}

function rollAll(): Item[] {
    // Seed one of every tier before rolling the rest, so the stash always shows
    // the full range - including the legendary two-sprite aura.
    const tiers = RARITY_ORDER.slice()
    while (tiers.length < NAMES.length) tiers.push(rollRarity(Math.random()))
    for (let i = tiers.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1))
        const t = tiers[i]; tiers[i] = tiers[j]; tiers[j] = t
    }
    return NAMES.map((name, i) => ({
        name,
        rarity: tiers[i],
        power: 12 + Math.floor(Math.random() * 88),
    }))
}

/**
 * The aura. Mounted only for tiers that have one; `deps: [rarity]` recreates the
 * system in place when an item is rerolled into a different tier.
 */
function RarityAura({ rarity }: { rarity: Rarity }) {
    const ref = useRef<VisualElement | null>(null)
    useParticles(ref, RARITY[rarity].aura!, [rarity])
    return (
        <View
            ref={ref}
            pickingMode="Ignore"
            style={{ position: "absolute", left: 0, top: 0, width: SLOT, height: SLOT }}
        />
    )
}

function Slot({ item, index, selected, dragging, onPick, onSelect, slotRef }: {
    item: Item
    index: number
    selected: boolean
    dragging: boolean
    onPick: (index: number, e: PointerEventData) => void
    onSelect: (index: number) => void
    slotRef: (el: VisualElement | null) => void
}) {
    const def = RARITY[item.rarity]
    const col = index % COLS
    const row = Math.floor(index / COLS)

    return (
        <View
            ref={slotRef}
            onPointerDown={(e) => { onSelect(index); onPick(index, e) }}
            style={{
                position: "absolute",
                left: col * (SLOT + GAP),
                top: row * (SLOT + GAP),
                width: SLOT,
                height: SLOT,
                opacity: dragging ? 0.25 : 1,
            }}
        >
            {/* chrome - kept off the particle host so its corners stay antialiased */}
            <View style={{
                position: "absolute",
                left: 0,
                top: 0,
                width: SLOT,
                height: SLOT,
                borderRadius: 10,
                borderWidth: selected ? 2 : 1,
                borderColor: selected ? def.color : "#242a38",
                backgroundColor: "#121622",
                alignItems: "center",
                justifyContent: "center",
            }}>
                <View style={{
                    width: 40,
                    height: 40,
                    borderRadius: 6,
                    backgroundColor: def.color,
                    opacity: 0.85,
                }} />
                <Text style={{ color: "#5d6577", fontSize: 10, marginTop: 8 }}>{item.power}</Text>
            </View>

            {def.aura && <RarityAura rarity={item.rarity} />}
        </View>
    )
}

export function Inventory() {
    const [items, setItems] = useState<Item[]>(rollAll)
    const [selected, setSelected] = useState(0)
    const [drag, setDrag] = useState<{ index: number; x: number; y: number } | null>(null)

    const overlayRef = useRef<VisualElement | null>(null)
    const slotEls = useRef<(VisualElement | null)[]>([])

    // Trail lives in panel space so it stays where it was spawned instead of
    // riding the emitter. One emitter per rarity: switching colors mid-drag is
    // then just an index, never a rebuild.
    const trail = useParticles(overlayRef, {
        max: 900,
        space: "panel",
        emitters: RARITY_ORDER.map((r) => ({
            rate: 0,
            shape: { type: "circle", radius: 5 },
            speed: [6, 34],
            lifetime: [0.35, 0.9],
            size: [4, 11],
            additiveness: 1,
            colorOverLife: [RARITY[r].color, RARITY[r].color + "00"],
            sizeOverLife: [1, 0.1],
        })),
    })

    function toLocal(e: PointerEventData) {
        const wb = overlayRef.current?.worldBound
        return wb ? { x: e.x - wb.x, y: e.y - wb.y } : { x: e.x, y: e.y }
    }

    function pick(index: number, e: PointerEventData) {
        const p = toLocal(e)
        setDrag({ index, x: p.x, y: p.y })
        const ei = RARITY_ORDER.indexOf(items[index].rarity)
        trail.emitters[ei]?.pos(p.x, p.y)
        trail.emitters[ei] && (trail.emitters[ei].rate = 150)
    }

    function move(e: PointerEventData) {
        if (!drag) return
        const p = toLocal(e)
        setDrag({ ...drag, x: p.x, y: p.y })
        const ei = RARITY_ORDER.indexOf(items[drag.index].rarity)
        trail.emitters[ei]?.pos(p.x, p.y) // one crossing per frame
    }

    function drop(e: PointerEventData) {
        if (!drag) return
        trail.emitters.forEach((em) => (em.rate = 0))

        // swap into whatever slot the pointer is over
        let target = -1
        for (let i = 0; i < slotEls.current.length; i++) {
            const wb = slotEls.current[i]?.worldBound
            if (wb && e.x >= wb.x && e.x <= wb.x + wb.width && e.y >= wb.y && e.y <= wb.y + wb.height) {
                target = i
                break
            }
        }
        if (target >= 0 && target !== drag.index) {
            const next = items.slice()
            const tmp = next[target]
            next[target] = next[drag.index]
            next[drag.index] = tmp
            setItems(next)
            setSelected(target)
        }
        setDrag(null)
    }

    const item = items[selected]
    const def = RARITY[item.rarity]

    return (
        <View
            style={{ flexGrow: 1, backgroundColor: "#080a10", alignItems: "center", justifyContent: "center" }}
            onPointerMove={move}
            onPointerUp={drop}
        >
            <View style={{ flexDirection: "row", alignItems: "flex-start" }}>
                {/* grid */}
                <View>
                    <View style={{ flexDirection: "row", justifyContent: "space-between", marginBottom: 12, width: GRID_W }}>
                        <Text style={{ color: "#e6e9f0", fontSize: 22, unityFontStyleAndWeight: "bold" }}>Stash</Text>
                        <View
                            onClick={() => { setItems(rollAll()); setSelected(0) }}
                            style={{
                                paddingTop: 6, paddingBottom: 6, paddingLeft: 14, paddingRight: 14,
                                borderRadius: 6, borderWidth: 1, borderColor: "#2b3346", backgroundColor: "#141a28",
                            }}
                        >
                            <Text style={{ color: "#9fb0cc", fontSize: 12 }}>Reroll stash</Text>
                        </View>
                    </View>

                    <View style={{ width: GRID_W, height: GRID_H }}>
                        {items.map((it, i) => (
                            <Slot
                                key={it.name}
                                item={it}
                                index={i}
                                selected={i === selected}
                                dragging={drag?.index === i}
                                onPick={pick}
                                onSelect={setSelected}
                                slotRef={(el) => { slotEls.current[i] = el }}
                            />
                        ))}
                    </View>

                    <Text style={{ color: "#39415a", fontSize: 12, marginTop: 12 }}>
                        Drag an item to swap slots — the trail stays in panel space. Reroll to rebuild every aura from its tier.
                    </Text>
                </View>

                {/* detail panel */}
                <View style={{
                    marginLeft: 26,
                    width: 240,
                    paddingTop: 18, paddingBottom: 18, paddingLeft: 18, paddingRight: 18,
                    borderRadius: 12,
                    borderWidth: 1,
                    borderColor: "#212838",
                    backgroundColor: "#0d1119",
                }}>
                    <Text style={{ color: def.color, fontSize: 12, unityFontStyleAndWeight: "bold" }}>
                        {def.label.toUpperCase()}
                    </Text>
                    <Text style={{ color: "#e6e9f0", fontSize: 20, marginTop: 6 }}>{item.name}</Text>
                    <Text style={{ color: "#6b7488", fontSize: 13, marginTop: 10 }}>{`Power ${item.power}`}</Text>
                    <View style={{ height: 1, backgroundColor: "#1c2231", marginTop: 14, marginBottom: 14 }} />
                    <Text style={{ color: "#586275", fontSize: 11 }}>
                        {def.aura
                            ? "Aura config comes straight from the rarity table."
                            : "Common items mount no system at all."}
                    </Text>
                </View>
            </View>

            {/* Drag trail + ghost. pickingMode Ignore so it never eats the drop. */}
            <View
                ref={overlayRef}
                pickingMode="Ignore"
                style={{ position: "absolute", left: 0, top: 0, right: 0, bottom: 0 }}
            >
                {drag && (
                    <View style={{
                        position: "absolute",
                        left: drag.x - 20,
                        top: drag.y - 20,
                        width: 40,
                        height: 40,
                        borderRadius: 6,
                        backgroundColor: RARITY[items[drag.index].rarity].color,
                    }} />
                )}
            </View>
        </View>
    )
}
